cmake_minimum_required(VERSION 3.24)
project(PSVR2BetaTests LANGUAGES CXX)
enable_testing()
add_executable(beta_tests beta_tests.cpp ../src/config/config_json.cpp)
target_compile_features(beta_tests PRIVATE cxx_std_20)
target_include_directories(beta_tests PRIVATE ../src/core ../src/config)
if(PSVR2_JSON_INCLUDE_DIR)
    target_include_directories(beta_tests PRIVATE "${PSVR2_JSON_INCLUDE_DIR}")
else()
    find_package(nlohmann_json CONFIG REQUIRED)
    target_link_libraries(beta_tests PRIVATE nlohmann_json::nlohmann_json)
endif()
add_test(NAME beta_geometry_and_config COMMAND beta_tests)
