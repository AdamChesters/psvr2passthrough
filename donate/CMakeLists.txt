# Compile the bundled content. No remote code or copy is downloaded at runtime.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/content.json" ADAMCH_CONTENT_JSON)
function(support_text output)
  string(JSON value GET "${ADAMCH_CONTENT_JSON}" ${ARGN})
  string(REPLACE "\\" "\\\\" value "${value}")
  string(REPLACE "\"" "\\\"" value "${value}")
  string(REPLACE "\n" "\\n" value "${value}")
  string(REPLACE "\r" "\\r" value "${value}")
  set(${output} "${value}" PARENT_SCOPE)
endfunction()
support_text(SUPPORT_TITLE title)
support_text(SUPPORT_INTRO intro)
support_text(SUPPORT_PROJECTS projectsText)
support_text(SUPPORT_DONATION_TITLE donationTitle)
support_text(SUPPORT_DONATION_TEXT donationText)
support_text(SUPPORT_DISCORD links discord)
support_text(SUPPORT_GITHUB links github)
support_text(SUPPORT_WEBSITE links website)
support_text(SUPPORT_DISCORD_LABEL actions discord)
support_text(SUPPORT_FEEDBACK_LABEL actions feedback)
support_text(SUPPORT_UPDATE_LABEL actions update)
support_text(SUPPORT_FEEDBACK_ENDPOINT feedback endpoint)
support_text(SUPPORT_FEEDBACK_TITLE feedback title)
support_text(SUPPORT_FEEDBACK_INTRO feedback intro)
support_text(SUPPORT_FEEDBACK_PRIVACY feedback privacy)
support_text(SUPPORT_FEEDBACK_SUCCESS feedback success)
support_text(SUPPORT_FEEDBACK_FAILURE feedback failure)
support_text(SUPPORT_FEEDBACK_SUBMIT feedback submit)
support_text(SUPPORT_FEEDBACK_PLACEHOLDER feedback placeholder)
string(JSON SUPPORT_FEEDBACK_MAX_BYTES GET "${ADAMCH_CONTENT_JSON}" feedback ack maxBytes)
string(JSON SUPPORT_FEEDBACK_STATUS GET "${ADAMCH_CONTENT_JSON}" feedback ack httpStatus)
support_text(SUPPORT_FEEDBACK_CONTENT_TYPE feedback ack contentType)
foreach(field RANGE 0 2)
  support_text(label feedback fields ${field} label)
  string(JSON maximum GET "${ADAMCH_CONTENT_JSON}" feedback fields ${field} maxLength)
  set(SUPPORT_FIELD_${field}_LABEL "${label}")
  set(SUPPORT_FIELD_${field}_MAX ${maximum})
endforeach()
string(JSON SUPPORT_PLATFORM_COUNT LENGTH "${ADAMCH_CONTENT_JSON}" platforms)
math(EXPR final_platform "${SUPPORT_PLATFORM_COUNT}-1")
set(SUPPORT_PLATFORMS "")
foreach(platform RANGE 0 ${final_platform})
  support_text(label platforms ${platform} label)
  support_text(url platforms ${platform} url)
  string(JSON notes_count LENGTH "${ADAMCH_CONTENT_JSON}" platforms ${platform} notes)
  math(EXPR final_note "${notes_count}-1")
  set(notes "")
  foreach(note RANGE 0 ${final_note})
    support_text(text platforms ${platform} notes ${note})
    if(NOT notes STREQUAL "")
      string(APPEND notes "\\n")
    endif()
    string(APPEND notes "${text}")
  endforeach()
  string(APPEND SUPPORT_PLATFORMS "    {\"${label}\", \"${url}\", \"${notes}\"},\n")
endforeach()
configure_file(support_content.hpp.in generated/support_content.hpp @ONLY)

target_include_directories(adamch_support PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(adamch_support PUBLIC imgui::imgui winhttp shell32)
target_compile_definitions(adamch_support PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
add_executable(support_ui_tests tests/support_ui_tests.cpp)
target_link_libraries(support_ui_tests PRIVATE adamch_support)
add_test(NAME bundled_support_ui COMMAND support_ui_tests)
